    83→function toPersianDigits(val: string | number): string {
    84→  const persianDigits = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
    85→  return String(val).replace(/\d/g, (d) => persianDigits[parseInt(d)]);
    86→}
    87→
    88→function formatTime(seconds: number): string {
    89→  const mins = Math.floor(seconds / 60);
    90→  const secs = seconds % 60;
    91→  return `${toPersianDigits(String(mins).padStart(2, '0'))}:${toPersianDigits(String(secs).padStart(2, '0'))}`;
    92→}
    93→
    94→function getTodayStr(): string {
    95→  const d = new Date();
    96→  const y = d.getFullYear();
    97→  const m = String(d.getMonth() + 1).padStart(2, '0');
    98→  const day = String(d.getDate()).padStart(2, '0');
    99→  return `${y}-${m}-${day}`;
   100→}
   101→
   102→function getDurationForMode(
   103→  mode: FocusType,
   104→  settings: { pomodoroDuration: number; shortBreakDuration: number; longBreakDuration: number }
   105→): number {
   106→  switch (mode) {
   107→    case 'pomodoro':
   108→      return settings.pomodoroDuration * 60;
   109→    case 'short-break':
   110→      return settings.shortBreakDuration * 60;
   111→    case 'long-break':
   112→      return settings.longBreakDuration * 60;
   113→  }
   114→}
   115→
   116→function formatFocusTime(totalMinutes: number): string {
   117→  if (totalMinutes < 60) {
   118→    return `${toPersianDigits(totalMinutes)} دقیقه`;
   119→  }
   120→  const hours = Math.floor(totalMinutes / 60);
   121→  const mins = totalMinutes % 60;
   122→  if (mins === 0) return `${toPersianDigits(hours)} ساعت`;
   123→  return `${toPersianDigits(hours)} ساعت و ${toPersianDigits(mins)} دقیقه`;
   124→}

   611→                      'h-16 w-16 rounded-full shadow-lg transition-all duration-300',

   629→                    variant="ghost"

   650→
   651→                {/* Session count */}

   654→                    {Array.from({ length: settings.longBreakInterval }).map((_, i) => (

   658→                          'h-2.5 w-2.5 rounded-full transition-all duration-300',
   659→                          i < sessionCount % settings.longBreakInterval || (sessionCount % settings.longBreakInterval === 0 && sessionCount > 0 && i < settings.longBreakInterval)

   664→                    ))}
   665→                    <span className="mr-2 text-xs text-muted-foreground">

   670→              </CardContent>
   671→            </Card>
   672→          </motion.div>
   673→
   674→          {/* Today's Sessions */}
   675→          <motion.div variants={itemVariants}>
   676→            <Card>
   677→              <CardHeader className="pb-3">
   678→                <div className="flex items-center justify-between">
   679→                  <CardTitle className="text-base">جلسات امروز</CardTitle>
   680→                  {todaySessions.length > 0 && (
   681→                    <Badge variant="secondary" className="gap-1">
   682→                      <Clock className="h-3 w-3" />
   683→                      مجموع تمرکز امروز: {formatFocusTime(todayTotalFocusMinutes)}
   684→                    </Badge>
   685→                  )}
   686→                </div>
   687→              </CardHeader>
   688→              <CardContent>
   689→                {todaySessions.length === 0 ? (
   690→                  <div className="flex flex-col items-center justify-center py-8 text-center text-muted-foreground">
   691→                    <CheckCircle2 className="mb-2 h-8 w-8 opacity-40" />

   697→                    {todaySessions.map((session, idx) => {

   699→                      const timeStr = toPersianDigits(
   700→                        `${String(sessionDate.getHours()).padStart(2, '0')}:${String(sessionDate.getMinutes()).padStart(2, '0')}`

   702→                      const typeLabel = MODE_LABELS[session.type];
   703→                      const typeIcon =
   704→                        session.type === 'pomodoro' ? (

   708→                        ) : (

   717→
   718→                      return (

   723→                          transition={{ delay: idx * 0.05 }}

   731→                            <span className="text-sm text-muted-foreground">
   732→                              {toPersianDigits(session.duration)} دقیقه
   733→                            </span>
   734→                          </div>
   735→                          <span className="text-xs text-muted-foreground">{timeStr}</span>
   736→                        </motion.div>
   737→                      );
   738→                    })}
   739→                  </div>
   740→                )}
   741→              </CardContent>
   742→            </Card>
   743→          </motion.div>
   744→        </motion.div>
   745→      </div>
   746→    </div>
