    85→function formatDateKey(date: Date): string {
    86→  const y = date.getFullYear();
    87→  const m = String(date.getMonth() + 1).padStart(2, '0');
    88→  const d = String(date.getDate()).padStart(2, '0');
    89→  return `${y}-${m}-${d}`;
    90→}
    91→
    92→function jalaaliDateKey(jd: JalaaliDate): string {
    93→  return `${jd.jy}/${String(jd.jm).padStart(2, '0')}/${String(jd.jd).padStart(2, '0')}`;
    94→}

    96→function getLast7Days(weekOffset: number): Date[] {
    97→  const days: Date[] = [];
    98→  const baseDate = new Date();
    99→  baseDate.setDate(baseDate.getDate() + weekOffset * 7);
   100→  for (let i = 6; i >= 0; i--) {
   101→    const date = new Date(baseDate);
   102→    date.setDate(date.getDate() - i);
   103→    date.setHours(0, 0, 0, 0);
   104→    days.push(date);
   105→  }
   106→  return days;
   107→}

   109→function isRecordedForDate(habit: Habit, date: Date): boolean {
   110→  if (!habit.records || habit.records.length === 0) return false;
   111→  const dateStr = formatDateKey(date);
   112→  return habit.records.some((r) => {
   113→    const recordDate = new Date(r.date);
   114→    return formatDateKey(recordDate) === dateStr;
   115→  });
   116→}
   117→
   118→function getRecordForDate(habit: Habit, date: Date) {
   119→  if (!habit.records || habit.records.length === 0) return null;
   120→  const dateStr = formatDateKey(date);
   121→  return habit.records.find((r) => {
   122→    const recordDate = new Date(r.date);
   123→    return formatDateKey(recordDate) === dateStr;
   124→  }) ?? null;
   125→}
   126→
   127→function calculateStreak(habit: Habit): number {
   128→  if (!habit.records || habit.records.length === 0) return 0;
   129→
   130→  const todayKey = formatDateKey(new Date());
   131→  const recordDates = habit.records
   132→    .map((r) => formatDateKey(new Date(r.date)))
   133→    .sort()
   134→    .reverse();
   135→
   136→  // Remove duplicates
   137→  const uniqueDates = [...new Set(recordDates)];
   138→
   139→  // Check if today is recorded; if not, check yesterday (streak might still be valid)
   140→  let startIndex = 0;
   141→  if (uniqueDates[0] !== todayKey) {
   142→    const yesterday = new Date();
   143→    yesterday.setDate(yesterday.getDate() - 1);
   144→    const yesterdayKey = formatDateKey(yesterday);
   145→    if (uniqueDates[0] !== yesterdayKey) return 0;
   146→  }
   147→
   148→  let streak = 0;
   149→  const checkDate = new Date();
   150→  if (uniqueDates[0] !== todayKey) {
   151→    checkDate.setDate(checkDate.getDate() - 1);
   152→  }
   153→
   154→  for (let i = 0; i < 365; i++) {
   155→    const key = formatDateKey(checkDate);
   156→    if (uniqueDates.includes(key)) {
   157→      streak++;
   158→      checkDate.setDate(checkDate.getDate() - 1);
   159→    } else {
   160→      break;
   161→    }
   162→  }
   163→
   164→  return streak;
   165→}
   166→
   167→function calculateCompletionRate(habit: Habit, last7Days: Date[]): number {
   168→  if (!habit.records || habit.records.length === 0) return 0;
   169→  const completed = last7Days.filter((d) => isRecordedForDate(habit, d)).length;
   170→  return Math.round((completed / 7) * 100);
   171→}

   173→function getWeekLabel(offset: number): string {
   174→  if (offset === 0) return 'هفته جاری';
   175→  if (offset === -1) return 'هفته قبل';
   176→  if (offset === 1) return 'هفته بعد';
   177→  if (offset < -1) return `${toPersianNum(Math.abs(offset))} هفته قبل`;
   178→  return `${toPersianNum(offset)} هفته بعد`;
   179→}
