   125→function calculateProgress(goal: Goal): number {
   126→  if (!goal.milestones || goal.milestones.length === 0) return 0;
   127→  const completed = goal.milestones.filter((m) => m.isCompleted).length;
   128→  return Math.round((completed / goal.milestones.length) * 100);
   129→}
   130→
   131→function formatGoalDate(dateStr?: string): string | null {
   132→  if (!dateStr) return null;
   133→  try {
   134→    const date = new Date(dateStr);
   135→    return formatJalaaliDateShort(date);
   136→  } catch {
   137→    return dateStr;
   138→  }
   139→}
   140→
   141→function generateId(): string {
   142→  return `ms_${Date.now()}_${Math.random().toString(36).substring(2, 9)}`;
   143→}

   369→      title: m.id === milestoneId ? m.title : m.title,

   378→      startDate: goal.startDate,

   384→  function addMilestone(goalId: string) {
   385→    const inputText = newMilestoneInputs[goalId]?.trim();
   386→    if (!inputText) return;
   387→
   388→    const goal = goals.find((g) => g.id === goalId);
   389→    if (!goal) return;
   390→

   392→      title: m.title,
   393→      order: m.order,
   394→      isCompleted: m.isCompleted,
   395→    }));
   396→
   397→    const newMilestones = [
   398→      ...existingMilestones,
   399→      {
   400→        title: inputText,
   401→        order: existingMilestones.length,
   402→        isCompleted: false,
   403→      },
   404→    ];
   405→
   406→    updateGoalMutation.mutate({
   407→      id: goal.id,
   408→      title: goal.title,
   409→      description: goal.description,
   410→      category: goal.category,
   411→      startDate: goal.startDate,
   412→      endDate: goal.endDate,
   413→      milestones: newMilestones,
   414→    });
   415→    setNewMilestoneInputs((prev) => ({ ...prev, [goalId]: '' }));
   416→  }
