    83→    83→function toPersianDigits(val: string | number): string {
    84→    84→  const persianDigits = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
    85→    85→  return String(val).replace(/\d/g, (d) => persianDigits[parseInt(d)]);
    86→    86→}

    88→    88→function formatTime(seconds: number): string {
    89→    89→  const mins = Math.floor(seconds / 60);
    90→    90→  const secs = seconds % 60;
    91→    91→  return `${toPersianDigits(String(mins).padStart(2, '0'))}:${toPersianDigits(String(secs).padStart(2, '0'))}`;
    92→    92→}
    93→    93→
    94→    94→function getTodayStr(): string {
    95→    95→  const d = new Date();
    96→    96→  const y = d.getFullYear();
    97→    97→  const m = String(d.getMonth() + 1).padStart(2, '0');
    98→    98→  const day = String(d.getDate()).padStart(2, '0');
    99→    99→  return `${y}-${m}-${day}`;
   100→   100→}

   102→   102→function getDurationForMode(
   103→   103→  mode: FocusType,
   104→   104→  settings: { pomodoroDuration: number; shortBreakDuration: number; longBreakDuration: number }
   105→   105→): number {
   106→   106→  switch (mode) {
   107→   107→    case 'pomodoro':
   108→   108→      return settings.pomodoroDuration * 60;
   109→   109→    case 'short-break':
   110→   110→      return settings.shortBreakDuration * 60;
   111→   111→    case 'long-break':
   112→   112→      return settings.longBreakDuration * 60;
   113→   113→  }
   114→   114→}

   116→   116→function formatFocusTime(totalMinutes: number): string {
   117→   117→  if (totalMinutes < 60) {
   118→   118→    return `${toPersianDigits(totalMinutes)} دقیقه`;
   119→   119→  }
   120→   120→  const hours = Math.floor(totalMinutes / 60);
   121→   121→  const mins = totalMinutes % 60;
   122→   122→  if (mins === 0) return `${toPersianDigits(hours)} ساعت`;
   123→   123→  return `${toPersianDigits(hours)} ساعت و ${toPersianDigits(mins)} دقیقه`;
   124→   124→}
