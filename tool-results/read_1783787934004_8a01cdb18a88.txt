   130→function getTodayStr(): string {
   131→  const d = new Date();
   132→  const y = d.getFullYear();
   133→  const m = String(d.getMonth() + 1).padStart(2, '0');
   134→  const day = String(d.getDate()).padStart(2, '0');
   135→  return `${y}-${m}-${day}`;
   136→}
   137→
   138→function getWeekStartStr(): string {
   139→  const d = new Date();
   140→  // Go back to Saturday (Persian week start)
   141→  const dayOfWeek = d.getDay(); // 0=Sun
   142→  const diff = (dayOfWeek + 1) % 7; // days since Saturday
   143→  d.setDate(d.getDate() - diff);
   144→  const y = d.getFullYear();
   145→  const m = String(d.getMonth() + 1).padStart(2, '0');
   146→  const day = String(d.getDate()).padStart(2, '0');
   147→  return `${y}-${m}-${day}`;
   148→}
   149→
   150→function formatDueDate(dateStr?: string, calendarType?: 'shamsi' | 'miladi'): string {
   151→  if (!dateStr) return '';
   152→  if (calendarType === 'shamsi') {
   153→    const parts = dateStr.split('-');
   154→    const d = new Date(Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]));
   155→    return formatJalaaliDateShort(d);
   156→  }
   157→  return dateStr;
   158→}

   497→function TaskDialog({
   498→  open,
   499→  onOpenChange,
   500→  editingTask,
   501→  projects,
   502→  labels,
   503→}: {
   504→  open: boolean;
   505→  onOpenChange: (open: boolean) => void;

   507→  projects: { id: string; title: string; color: string }[];
   508→  labels: { id: string; name: string; color: string }[];
   509→}) {
   510→  const queryClient = useQueryClient();
   511→  const isEditing = !!editingTask;
   512→
   513→  const [form, setForm] = useState({
   514→    title: '',
   515→    description: '',
   516→    priority: 'medium' as TaskPriority,
   517→    status: 'todo' as TaskStatus,
   518→    dueDate: '',
   519→    dueTime: '',
   520→    projectId: '',
   521→    labelId: '',
   522→    estimatedMinutes: '',
   523→  });
   524→
   525→  // Reset form when dialog opens or editingTask changes
   526→  const handleOpenChange = useCallback(
   527→    (nextOpen: boolean) => {
   528→      if (nextOpen && editingTask) {
   529→        setForm({
   530→          title: editingTask.title,
   531→          description: editingTask.description || '',
   532→          priority: editingTask.priority,
   533→          status: editingTask.status,
   534→          dueDate: editingTask.dueDate || '',
   535→          dueTime: editingTask.dueTime || '',
   536→          projectId: editingTask.projectId || '',
   537→          labelId: editingTask.labelId || '',
   538→          estimatedMinutes: editingTask.estimatedMinutes
   539→            ? String(editingTask.estimatedMinutes)
   540→            : '',
   541→        });
   542→      } else if (nextOpen && !editingTask) {
   543→        setForm({
   544→          title: '',
   545→          description: '',
   546→          priority: 'medium',
   547→          status: 'todo',
   548→          dueDate: '',
   549→          dueTime: '',
   550→          projectId: '',
   551→          labelId: '',
   552→          estimatedMinutes: '',
   553→        });
   554→      }
   555→      onOpenChange(nextOpen);
   556→    },
   557→    [editingTask, onOpenChange]
   558→  );
   559→
   560→  const updateField = <K extends keyof typeof form>(key: K, value: (typeof form)[K]) => {
   561→    setForm((prev) => ({ ...prev, [key]: value }));
   562→  };
   563→
   564→  const createMutation = useMutation({
   565→    mutationFn: async (data: Record<string, unknown>) => {
   566→      const res = await fetch('/api/tasks', {
   567→        method: 'POST',
   568→        headers: { 'Content-Type': 'application/json' },
   569→        body: JSON.stringify({
   570→          ...data,
   571→          estimatedMinutes: data.estimatedMinutes
   572→            ? Number(data.estimatedMinutes)
   573→            : null,
   574→        }),
   575→      });
   576→      if (!res.ok) throw new Error('خطا در ایجاد وظیفه');
   577→      return res.json();
   578→    },
   579→    onSuccess: () => {
   580→      queryClient.invalidateQueries({ queryKey: ['tasks'] });
   581→      handleOpenChange(false);
   582→    },
   583→  });
   584→
   585→  const updateMutation = useMutation({
   586→    mutationFn: async (data: Record<string, unknown>) => {
   587→      const res = await fetch('/api/tasks', {
   588→        method: 'PUT',
   589→        headers: { 'Content-Type': 'application/json' },
   590→        body: JSON.stringify({
   591→          ...data,
   592→          id: editingTask!.id,
   593→          estimatedMinutes: data.estimatedMinutes
   594→            ? Number(data.estimatedMinutes)
   595→            : null,
   596→          projectId: data.projectId || null,
   597→          labelId: data.labelId || null,
   598→        }),
   599→      });
   600→      if (!res.ok) throw new Error('خطا در بروزرسانی وظیفه');
   601→      return res.json();
   602→    },
   603→    onSuccess: () => {
   604→      queryClient.invalidateQueries({ queryKey: ['tasks'] });
   605→      handleOpenChange(false);
   606→    },
   607→  });
   608→
   609→  const handleSubmit = (e: React.FormEvent) => {
   610→    e.preventDefault();
   611→    if (!form.title.trim()) return;
   612→
   613→    const payload = {
   614→      title: form.title.trim(),
   615→      description: form.description.trim() || null,
   616→      priority: form.priority,
   617→      status: form.status,
   618→      dueDate: form.dueDate || null,
   619→      dueTime: form.dueTime || null,

   821→  const [dialogOpen, setDialogOpen] = useState(false);
   822→  const [editingTask, setEditingTask] = useState<Task | null>(null);

   825→  /* ---------- sensors for DnD ---------- */
   826→  const sensors = useSensors(
   827→    useSensor(PointerSensor, {
   828→      activationConstraint: { distance: 5 },
   829→    })
   830→  );
   831→
   832→  /* ---------- filtered tasks ---------- */
   833→  const filteredTasks = useMemo(() => {
   834→    let result = tasks.filter((t) => !t.isArchived);
   835→
   836→    const todayStr = getTodayStr();
   837→    const weekStartStr = getWeekStartStr();
   838→
   839→    switch (filter) {
   840→      case 'today':
   841→        result = result.filter((t) => t.dueDate === todayStr);
   842→        break;
   843→      case 'week':
   844→        result = result.filter(
   845→          (t) => t.dueDate && t.dueDate >= weekStartStr && t.dueDate <= todayStr
   846→        );
   847→        break;
   848→      case 'important':
   849→        result = result.filter((t) => t.priority === 'high' || t.priority === 'urgent');
   850→        break;
   851→      case 'urgent':
   852→        result = result.filter((t) => t.priority === 'urgent');
   853→        break;
   854→    }
   855→
   856→    return result;
   857→  }, [tasks, filter]);
   858→
   859→  /* ---------- tasks by status ---------- */
   860→  const tasksByStatus = useMemo(() => {
   861→    const map: Record<TaskStatus, Task[]> = {
   862→      todo: [],
   863→      'in-progress': [],
   864→      done: [],
   865→      archived: [],
   866→    };
   867→    for (const t of filteredTasks) {
   868→      map[t.status]?.push(t);
   869→    }
   870→    return map;
   871→  }, [filteredTasks]);
   872→
   873→  /* ---------- list view sorted IDs ---------- */
   874→  const listTaskIds = useMemo(
   875→    () => filteredTasks.map((t) => t.id),
   876→    [filteredTasks]
   877→  );
   878→
   879→  /* ---------- mutations ---------- */
   880→  const toggleDoneMutation = useMutation({
   881→    mutationFn: async ({ id, status }: { id: string; status: string }) => {
   882→      const res = await fetch('/api/tasks', {
   883→        method: 'PUT',
   884→        headers: { 'Content-Type': 'application/json' },
   885→        body: JSON.stringify({ id, status }),
   886→      });
   887→      return res.json();
   888→    },
   889→    onSuccess: () => {
   890→      queryClient.invalidateQueries({ queryKey: ['tasks'] });
   891→    },
   892→  });
   893→
   894→  const deleteMutation = useMutation({
   895→    mutationFn: async (id: string) => {
   896→      const res = await fetch(`/api/tasks?id=${id}`, { method: 'DELETE' });
   897→      if (!res.ok) throw new Error('خطا در حذف');
   898→      return res.json();
   899→    },
   900→    onSuccess: () => {
   901→      queryClient.invalidateQueries({ queryKey: ['tasks'] });
   902→    },
   903→  });
   904→
   905→  const archiveMutation = useMutation({
   906→    mutationFn: async (id: string) => {
   907→      const res = await fetch('/api/tasks', {
   908→        method: 'PUT',
   909→        headers: { 'Content-Type': 'application/json' },
   910→        body: JSON.stringify({ id, isArchived: true }),
   911→      });
   912→      return res.json();
   913→    },
   914→    onSuccess: () => {
   915→      queryClient.invalidateQueries({ queryKey: ['tasks'] });
   916→    },
   917→  });
   918→
   919→  const reorderMutation = useMutation({
   920→    mutationFn: async (reorderedTasks: { id: string; order: number }[]) => {
   921→      await Promise.all(
   922→        reorderedTasks.map((t) =>
   923→          fetch('/api/tasks', {
   924→            method: 'PUT',
   925→            headers: { 'Content-Type': 'application/json' },
   926→            body: JSON.stringify({ id: t.id, order: t.order }),
   927→          })
   928→        )
   929→      );
   930→    },
   931→    onSuccess: () => {
   932→      queryClient.invalidateQueries({ queryKey: ['tasks'] });
   933→    },
   934→  });
   935→
   936→  /* ---------- handlers ---------- */
   937→  const handleToggleDone = (id: string, done: boolean) => {

   948→  };

   967→

   969→    const newIndex = listTaskIds.indexOf(over.id as string);

  1057→        <motion.div

  1067→            با کلیک روی دکمه زیر اولین وظیفه خود را ایجاد کنید

  1172→            </div>

  1174→        </DndContext>

  1176→
  1177→      {/* Task Dialog */}
  1178→      <TaskDialog
